#!/bin/bash
# Usage: pair-under-load.sh <xctestrun> <result-name> <load-processes> <only-testing>...
# Runs inside the caller's lockf hold; the CPU load lives only for this xcodebuild.
set -u
run="$1"; name="$2"; load="$3"; shift 3
W=~/.claude/orchestrate/cuadrao-iphone-candidate/w1h
pids=()
[ "$load" -gt 0 ] && for _ in $(seq 1 "$load"); do yes > /dev/null & pids+=($!); done
args=()
for t in "$@"; do args+=("-only-testing:ArgusFoundationUITests/$t"); done
xcodebuild test-without-building -xctestrun "$run" \
  -destination "platform=iOS Simulator,id=8AFB6084-8918-416E-9164-E21061306BEC" \
  -resultBundlePath "$W/$name.xcresult" -collect-test-diagnostics never \
  -parallel-testing-enabled NO "${args[@]}" > "$W/$name.log" 2>&1
code=$?
for p in ${pids[@]+"${pids[@]}"}; do kill "$p"; done
echo "$name exit $code load $load"
grep -E "Test Case .*(passed|failed)|error:" "$W/$name.log"
